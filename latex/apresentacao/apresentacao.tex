\section{Introdução}
% =====================================================================

% Slide: Contextualização
\begin{frame}{Contextualização: equipamentos ``caixa preta''}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Situação real:}
        \begin{itemize}
            \item Muitos equipamentos operam como ``caixas pretas''.
            \item O estado real (ligado/desligado) não chega à infraestrutura IoT.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Consequências:}
        \begin{itemize}
            \item Ciclo operacional desconhecido.
            \item Indicadores de produção e manutenção imprecisos.
            \item Soluções tradicionais não escalam para a diversidade de plantas.
        \end{itemize}
    \end{columns}
\end{frame}

% Slide: Por que não limiares fixos?
\begin{frame}{Por que não simplesmente definir limiares fixos?}
    \textbf{Ideia ingênua:} vibração $\approx 0$ e corrente $\approx 0$ $\Rightarrow$ DESLIGADO.

    \vspace{0.2cm}
    \textbf{Problemas na prática:}
    \begin{itemize}
        \item \textbf{Cada equipamento é único:} uma motobomba de 125\,cv vibra a $\sim$8\,mm/s em operação; um mancal opera abaixo de 1\,mm/s. O mesmo limiar fixo classificaria ambos errado.
        \item \textbf{Desgaste natural:} após 1 ano de uso o mesmo equipamento, em plenas condições, pode vibrar 20\% a mais. Um limiar fixo começa a classificar estado LIGADO como DESLIGADO.
        \item \textbf{Ambiente não controlado:} vibração de máquinas vizinhas, sazonalidade térmica e interferências eletromagnéticas sobrepõem-se ao sinal real, deslocando a ``linha de base''.
    \end{itemize}

\end{frame}

% Slide: Objetivo
\begin{frame}{Objetivo}
    \small
    \begin{block}{Meta principal}
        Desenvolver um sistema \textbf{não invasivo} e \textbf{automático} capaz de detectar os estados LIGADO/DESLIGADO de equipamentos industriais, utilizando \textbf{exclusivamente} os sensores IoT já instalados na planta --- sem necessidade de limiares manuais, rotulação humana ou hardware adicional.
    \end{block}
    \vspace{0.2cm}
    \textbf{Diferenciais da solução:}
    \begin{itemize}
        \item \textbf{Agnóstico ao tipo:} um único sistema atende equipamentos elétricos (com corrente e RPM) e mecânicos (apenas vibração e temperatura), sem reconfiguração.
        \item \textbf{Não supervisionado:} o algoritmo aprende os padrões operacionais de cada equipamento diretamente dos dados (sem necessidade de exemplos rotulados).
        \item \textbf{Baixo custo e escalável:} reaproveita a infraestrutura IoT existente e processa cada equipamento em menos de 14 minutos.
    \end{itemize}
\end{frame}

% Slide: Lição metodológica (CNN -> K-Means)
\begin{frame}{Por que aprendizado não supervisionado?}
    \textbf{Primeira tentativa: redes profundas (CNN / \textit{Autoencoders}).}
    \begin{itemize}
        \item Acurácia de treino de 100\%, mas os modelos \textbf{memorizavam} os padrões do conjunto de treino.
        \item \textbf{Extremamente sensíveis a variações bruscas:} pequenas mudanças nos valores de corrente ou temperatura causavam predições completamente diferentes (o modelo não entendia o fenômeno, apenas decorava sequências).
    \end{itemize}
    \vspace{0.2cm}
    \begin{block}{Nova solução}
        Complexidade excessiva para um problema binário. Foi implementado o \textbf{K-Means} puro: robusto a variações, interpretável, sem \textit{overfitting} e de treino rápido.
    \end{block}
\end{frame}

% Slide: Tecnologias
\begin{frame}{Tecnologias e ferramentas}
    \begin{itemize}
        \item \textbf{Linguagem:} Python 3.11+
        \item \textbf{Dados:} \textbf{InfluxDB} (séries temporais para IoT)
        \item \textbf{Processamento:} \textit{Pandas}, \textit{NumPy}, \textit{SciPy} (interpolação PCHIP/KNN)
        \item \textbf{\textit{Machine Learning}:} \textit{Scikit-learn} (K-Means, MinMax, PCA)
        \item \textbf{Interface:} \textit{CustomTkinter} (GUI para operadores)
        \item \textbf{Visualização:} \textit{Matplotlib} (gráficos 3D), \textit{Plotly} (web)
    \end{itemize}
\end{frame}

% Slide: Coleta e segmentação
\begin{frame}{Coleta e segmentação temporal}
    \begin{itemize}
        \item \textbf{3 fontes (elétrico):} \texttt{validated\_default} (20\,s), \texttt{estimated} ($\sim$1\,min) e \texttt{validated\_slip} (2\,min). Mecânicos usam apenas 2.
        \item \textbf{Download} em \texttt{.csv} via \textit{script} Python a partir do InfluxDB.
        \item \textbf{Segmentação por lacunas:} \textit{gaps} $>$ 3\,h $\Rightarrow$ novos períodos independentes.
        \item Cada período válido recebe uma \textbf{linha temporal uniforme} (grade de 20\,em\,20\,s).
    \end{itemize}
    \vspace{0.2cm}
    \begin{block}{Tratamento das lacunas internas}
        Intervalos sem dados dentro de um período ($\leq$ 3\,h) são preenchidos por \textbf{interpolação adaptativa}: \textit{spline} hermitiana (PCHIP) para lacunas curtas ou KNN temporal para lacunas médias, garantindo continuidade na série.
    \end{block}
\end{frame}

% Slide: Sincronização e fusão das tabelas
\begin{frame}{Sincronização e fusão das tabelas de dados}
    \textbf{Dados iniciais --- 3 tabelas com frequências distintas (elétrico):}
    \vspace{0.1cm}

    \footnotesize
    \begin{tabular}{lll}
        \hline
        \textbf{Fonte} & \textbf{Frequência} & \textbf{Atributos principais} \\
        \hline
        \texttt{validated\_default} & 20\,s & magnetômetro, temperatura, vibração RMS \\
        \texttt{estimated}         & $\sim$1\,min & corrente, RPM, vibração estimada \\
        \texttt{validated\_slip}   & 2\,min & frequência de escorregamento, harmônicas \\
        \hline
    \end{tabular}
    \normalsize

    \vspace{0.3cm}
    \textbf{Processo de fusão:}
    \begin{enumerate}
        \item \texttt{validated\_default} (20\,s) é a \textbf{base temporal} de referência.
        \item \texttt{estimated} é alinhada via \textit{backward-fill} (tolerância de 6\,min).
        \item \texttt{validated\_slip} é alinhada via \textit{backward-fill} (tolerância de 3\,min).
        \item Exigência de \textbf{$>$ 70\%} de sobreposição temporal entre fontes.
    \end{enumerate}

    \begin{block}{\textit{Backward-fill}}
        Cada instante sem leitura recebe o \textbf{próximo valor futuro} disponível dentro da janela de tolerância, garantindo que só dados já existentes preencham o grid, sem extrapolar valores passados.
    \end{block}
\end{frame}

% Slide: Classificação ponderada (scores)
\begin{frame}{Classificação ponderada por física}
    \textbf{Centróides:} cada \textit{cluster} $k$ do K-Means possui um centróide $\bar{C}_k$, que é o ponto médio (em cada dimensão) de todas as amostras atribuídas àquele grupo. O centróide resume o ``perfil típico'' de operação daquele grupo.

    \vspace{0.2cm}
    \textbf{\textit{Score} ponderado de cada centróide:}
    \begin{align*}
        S_{\text{elétrico}}  &= 2{,}0\,\bar{C}_{\text{corr}} + 2{,}0\,\bar{C}_{\text{RPM}} + 1{,}0\,\bar{C}_{\text{vib}} + 0{,}5\,\bar{C}_{\text{temp}} \\
        S_{\text{mecânico}} &= 1{,}5\,\bar{C}_{\text{temp}} + 1{,}0\,\bar{C}_{\text{vib}} + 0{,}5\,\bar{C}_{\text{mag}}
    \end{align*}
    \begin{itemize}
        \item Os \textit{clusters} são ordenados pelo \textit{score}: os \textbf{dois de menor valor} são avaliados como candidatos ao estado \textbf{DESLIGADO} (perfis de baixa atividade), passando em seguida pela etapa de validação.
    \end{itemize}
\end{frame}

% Slide: 1 ou 2 clusters DESLIGADO (movido para logo apos a classificacao ponderada)
\begin{frame}{Dois candidatos DESLIGADO: design e validação}
    \textbf{Design intencional:} os dois \textit{clusters} de menor \textit{score} são \textit{sempre} avaliados como candidatos ao estado DESLIGADO, em todos os equipamentos.

    \vspace{0.2cm}
    \textbf{Etapa de validação (aplicada a todos):}
    \begin{itemize}
        \item Diferença de \textit{score} entre os dois candidatos $<$ 0,3.
        \item Ambos os \textit{scores} $<$ 0,5 (baixa atividade absoluta).
        \item Vibração do 2\textdegree{} candidato $<$ 1,5 (nível residual).
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Resultado após validação:}
    \begin{itemize}
        \item \textbf{Na maioria dos casos:} o 2\textdegree{} candidato falha (regime de baixa carga, não repouso) $\Rightarrow$ reclassificado como LIGADO.
        \item \textbf{Evento raro:} ambos sobrevivem $\Rightarrow$ dois perfis distintos de inatividade real (ex: temperatura ambiente alta vs.\ baixa).
    \end{itemize}

\end{frame}

\begin{frame}{Impactos socioambientais}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Impacto ambiental:}
        \begin{itemize}
            \item \textbf{Gestão energética:} identificação precisa de períodos ociosos revela desperdício oculto (equipamentos ligados desnecessariamente).

        \column{0.5\textwidth}
        \textbf{Impacto social:}
        \begin{itemize}
            \item \textbf{Segurança operacional:} monitoramento contínuo antecipa paradas, reduzindo falhas inesperadas que colocam trabalhadores em risco.
            \item \textbf{Democratização:} GUI permite que operadores sem conhecimento de programação treinem modelos e analisem dados.
            \item \textbf{Baixo custo de adoção:} funciona com sensores já instalados, viável para pequenas e médias indústrias.
        \end{itemize}
    \end{columns}
\end{frame}
