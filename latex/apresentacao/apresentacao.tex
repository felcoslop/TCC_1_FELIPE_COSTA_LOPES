\section{Introdução}
% =====================================================================

% Slide: Contextualização
\begin{frame}{Contextualização: equipamentos ``caixa preta''}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Situação real:}
        \begin{itemize}
            \item Muitos equipamentos operam como ``caixas pretas''.
            \item O estado real (ligado/desligado) não chega à infraestrutura IoT.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Consequências:}
        \begin{itemize}
            \item Ciclo operacional desconhecido.
            \item Indicadores de produção e manutenção imprecisos.
            \item Soluções tradicionais não escalam para a diversidade de plantas.
        \end{itemize}
    \end{columns}
\end{frame}

% Slide: Objetivo
\begin{frame}{Objetivo}
    \begin{block}{Meta principal}
        Detectar de forma \textbf{não invasiva} e \textbf{automática} os estados LIGADO/DESLIGADO usando sensores IoT \textbf{já instalados}.
    \end{block}
    \vspace{0.3cm}
    \textbf{Diferenciais:}
    \begin{itemize}
        \item \textbf{Agnóstico:} atende equipamentos elétricos e mecânicos.
        \item \textbf{Não supervisionado:} aprende os padrões sem rotulação manual.
        \item \textbf{Baixo custo:} reaproveita a infraestrutura IoT existente.
    \end{itemize}
\end{frame}

% Slide: Lição metodológica (CNN -> K-Means)
\begin{frame}{Por que aprendizado não supervisionado?}
    \textbf{Primeira tentativa: redes profundas (CNN / \textit{Autoencoders}).}
    \begin{itemize}
        \item Acurácia de treino aparente de 100\%, mas \textbf{não generalizavam}.
        \item Classificavam como LIGADO períodos com RPM\,=\,0 e corrente nula.
    \end{itemize}
    \vspace{0.3cm}
    \begin{block}{Lição}
        Problemas simples pedem soluções simples. Migrou-se para \textbf{K-Means}: robusto, interpretável, sem \textit{overfitting} e de treino rápido.
    \end{block}
\end{frame}

% Slide: Tecnologias
\begin{frame}{Tecnologias e ferramentas}
    \begin{itemize}
        \item \textbf{Linguagem:} Python 3.11+
        \item \textbf{Dados:} \textbf{InfluxDB} (séries temporais para IoT)
        \item \textbf{Processamento:} \textit{Pandas}, \textit{NumPy}, \textit{SciPy} (interpolação PCHIP/KNN)
        \item \textbf{\textit{Machine Learning}:} \textit{Scikit-learn} (K-Means, MinMax, PCA)
        \item \textbf{Interface:} \textit{CustomTkinter} (GUI para operadores)
        \item \textbf{Visualização:} \textit{Matplotlib} (gráficos 3D), \textit{Plotly} (web)
    \end{itemize}
\end{frame}

% Slide: Visão geral do pipeline
\begin{frame}{Visão geral do \textit{pipeline}}
    \begin{columns}
        \column{0.46\textwidth}
        \textbf{Pipeline dual e modular:}
        \begin{enumerate}
            \item Detecção automática do tipo (elétrico/mecânico).
            \item Coleta e segmentação.
            \item Interpolação e tratamento de \textit{outliers}.
            \item Sincronização e normalização.
            \item Classificação K-Means.
        \end{enumerate}
        \column{0.54\textwidth}
        \begin{center}
            \includegraphics[height=0.78\textheight]{05_pipeline_treino.png}
        \end{center}
    \end{columns}
\end{frame}

% Slide: Coleta e segmentação
\begin{frame}{Coleta e segmentação temporal}
    \begin{itemize}
        \item \textbf{3 fontes (elétrico):} \texttt{validated\_default} (20\,s), \texttt{estimated} ($\sim$1\,min) e \texttt{validated\_slip} (2\,min). Mecânicos usam apenas 2.
        \item \textbf{Download} em \texttt{.csv} via \textit{script} Python a partir do InfluxDB.
        \item \textbf{Segmentação por lacunas:} \textit{gaps} $>$ 3\,h $\Rightarrow$ novos períodos independentes.
        \item Cada período válido recebe uma \textbf{linha temporal uniforme} (grade de 20\,em\,20\,s).
    \end{itemize}
    \vspace{0.2cm}
    \begin{block}{Tratamento das lacunas internas}
        Intervalos sem dados dentro de um período ($\leq$ 3\,h) são preenchidos por \textbf{interpolação adaptativa}: \textit{spline} hermitiana (PCHIP) para lacunas curtas ou KNN temporal para lacunas médias, garantindo continuidade na série.
    \end{block}
\end{frame}

% Slide: Sincronização e fusão das tabelas
\begin{frame}{Sincronização e fusão das tabelas de dados}
    \textbf{Dados iniciais --- 3 tabelas com frequências distintas (elétrico):}
    \vspace{0.1cm}

    \footnotesize
    \begin{tabular}{lll}
        \hline
        \textbf{Fonte} & \textbf{Frequência} & \textbf{Atributos principais} \\
        \hline
        \texttt{validated\_default} & 20\,s & magnetômetro, temperatura, vibração RMS \\
        \texttt{estimated}         & $\sim$1\,min & corrente, RPM, vibração estimada \\
        \texttt{validated\_slip}   & 2\,min & frequência de escorregamento, harmônicas \\
        \hline
    \end{tabular}
    \normalsize

    \vspace{0.3cm}
    \textbf{Processo de fusão:}
    \begin{enumerate}
        \item \texttt{validated\_default} (20\,s) é a \textbf{base temporal} de referência.
        \item \texttt{estimated} é alinhada via \textit{backward-fill} (tolerância de 6\,min).
        \item \texttt{validated\_slip} é alinhada via \textit{backward-fill} (tolerância de 3\,min).
        \item Exigência de \textbf{$>$ 70\%} de sobreposição temporal entre fontes.
    \end{enumerate}

    \vspace{0.1cm}
    \begin{block}{Resultado}
        Tabela unificada com \textbf{24 atributos}, todos a 20\,s, com 100\% de cobertura. Ex.: $c_{636}$ gera 863.450 registros sincronizados.
    \end{block}
\end{frame}

% Slide: Interpolação adaptativa
\begin{frame}{Interpolação adaptativa de lacunas}
    \begin{columns}
        \column{0.42\textwidth}
        \textbf{Estratégia por tamanho:}
        \begin{itemize}
            \item $\leq$ 30\,min: \textbf{PCHIP} (sem oscilações de Runge).
            \item 30\,min--3\,h: \textbf{KNN} temporal + linear.
            \item $>$ 3\,h: \textbf{não interpola} (novo período).
        \end{itemize}
        \vspace{0.2cm}
        \footnotesize LSTM testada, mas sem ganho frente ao KNN --- complexidade desnecessária.
        \column{0.58\textwidth}
        \begin{center}
            \includegraphics[width=\linewidth]{02_comparativo_knn_lstm_subplots.png}
        \end{center}
    \end{columns}
\end{frame}

% Slide: Tratamento de outliers
\begin{frame}{Tratamento de \textit{outliers} (IQR)}
    \begin{columns}
        \column{0.42\textwidth}
        \begin{itemize}
            \item IQR com multiplicador conservador (3,0).
            \item Proteção de inércia: sequências $\geq$ 10 amostras são \textbf{preservadas} (não são ruído).
            \item Filtro anti-EMI: zera RPM/corrente espúrios (\textless\,2\,A).
        \end{itemize}
        \column{0.58\textwidth}
        \begin{center}
            \includegraphics[width=\linewidth]{04_remocao_outliers.png}
        \end{center}
    \end{columns}
\end{frame}

% Slide: Normalização
\begin{frame}{Normalização robusta}
    \begin{columns}
        \column{0.42\textwidth}
        \begin{itemize}
            \item \textit{MinMax Scaler} para o intervalo $[0,1]$.
            \item \textit{Clipping} de \textit{outliers} no percentil 99,5\%.
            \item Preserva a forma e a distribuição; ajusta apenas a escala.
            \item Equilibra grandezas de escalas distintas.
        \end{itemize}
        \column{0.58\textwidth}
        \begin{center}
            \includegraphics[width=\linewidth]{01_minmax_scaler.png}
        \end{center}
    \end{columns}
\end{frame}

% Slide: K-Means
\begin{frame}{O coração do sistema: K-Means}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Por que não um limiar fixo?}
        \begin{itemize}
            \item Vibração residual de máquinas vizinhas.
            \item Variação da temperatura ambiente.
        \end{itemize}
        \vspace{0.2cm}
        \textbf{Configuração:}
        \begin{itemize}
            \item \textbf{k = 6} (proporção $\sim$1:2 entre desligado e ligado, captando transições).
            \item \textit{score} ponderado pela física.
        \end{itemize}

% Slide: Classificação ponderada (scores)
\begin{frame}{Classificação ponderada por física}
    \textbf{Centróides:} cada \textit{cluster} $k$ do K-Means possui um centróide $\bar{C}_k$, que é o ponto médio (em cada dimensão) de todas as amostras atribuídas àquele grupo. O centróide resume o ``perfil típico'' de operação daquele grupo.

    \vspace{0.2cm}
    \textbf{\textit{Score} ponderado de cada centróide:}
    \begin{align*}
        S_{\text{elétrico}}  &= 2{,}0\,\bar{C}_{\text{corr}} + 2{,}0\,\bar{C}_{\text{RPM}} + 1{,}0\,\bar{C}_{\text{vib}} + 0{,}5\,\bar{C}_{\text{temp}} \\
        S_{\text{mecânico}} &= 1{,}5\,\bar{C}_{\text{temp}} + 1{,}0\,\bar{C}_{\text{vib}} + 0{,}5\,\bar{C}_{\text{mag}}
    \end{align*}
    \begin{itemize}
        \item Os \textit{clusters} são ordenados pelo \textit{score}: o de menor valor é classificado como \textbf{DESLIGADO} (perfil de baixa atividade).
        \item \textbf{Travas físicas:} corrente $<$ 15\% e RPM $<$ 20\% $\Rightarrow$ DESLIGADO forçado.
    \end{itemize}
\end{frame}

% Slide: 1 ou 2 clusters DESLIGADO
\begin{frame}{Quando o K-Means encontra 2 \textit{clusters} DESLIGADO}
    \textbf{Caso padrão (maioria):} apenas 1 dos 6 \textit{clusters} é classificado como DESLIGADO --- o de menor \textit{score}.

    \vspace{0.2cm}
    \textbf{Caso raro --- 2 \textit{clusters} DESLIGADO:}
    \begin{itemize}
        \item Ocorre quando o equipamento apresenta \textbf{dois perfis distintos de inatividade}: por exemplo, desligamento com temperatura ambiente \textbf{alta} (verão) vs. \textbf{baixa} (inverno), gerando centróides separados.
        \item O sistema admite um segundo \textit{cluster} DESLIGADO \textbf{somente} se: diferença de \textit{score} $<$ 0,3, ambos \textit{scores} $<$ 0,5 e vibração $<$ 1,5 (residual).
    \end{itemize}

    \vspace{0.2cm}
    \begin{block}{Exemplo real: equipamento $c_{637}$}
        Na visualização RPM $\times$ Vibração $\times$ Tempo, identificam-se \textbf{dois agrupamentos DESLIGADO} distintos e múltiplos \textit{clusters} LIGADO, refletindo variações sazonais no perfil de parada.
    \end{block}
\end{frame}

% Slide: Impactos socioambientais
\begin{frame}{Impactos socioambientais}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Impacto ambiental:}
        \begin{itemize}
            \item \textbf{Gestão energética:} identificação precisa de períodos ociosos revela desperdício oculto (equipamentos ligados desnecessariamente).
            \item \textbf{Redução de resíduos:} manutenção baseada em horas reais evita troca prematura de peças, óleos e lubrificantes.
            \item \textbf{Sobriedade computacional:} K-Means treina em minutos em hardware modesto, sem GPU --- menor consumo energético que redes profundas.
        \end{itemize}
        \column{0.5\textwidth}
        \textbf{Impacto social:}
        \begin{itemize}
            \item \textbf{Segurança operacional:} monitoramento contínuo antecipa paradas, reduzindo falhas inesperadas que colocam trabalhadores em risco.
            \item \textbf{Democratização:} GUI permite que operadores sem conhecimento de programação treinem modelos e analisem dados.
            \item \textbf{Baixo custo de adoção:} funciona com sensores já instalados, viável para pequenas e médias indústrias.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Conclusão e próximos passos}

    \vspace{0.3cm}
    \textbf{Trabalho futuro principal:}

    \end{itemize}
