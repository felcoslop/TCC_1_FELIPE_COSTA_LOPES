\begin{description}
    \item[ARIMA] \textit{AutoRegressive Integrated Moving Average} (Média Móvel Integrada Autoregressiva)
    \item[CNN] \textit{Convolutional Neural Network} (Rede Neural Convolucional)
    \item[CSV] \textit{Comma-Separated Values} (Valores Separados por Vírgula)
    \item[EMI] \textit{Electromagnetic Interference} (Interferência Eletromagnética)
    \item[GB] \textit{Gigabyte}
    \item[GIF] \textit{Graphics Interchange Format} (Formato de Intercâmbio de Gráficos)

    \item[GUI] \textit{Graphical User Interface} (Interface Gráfica do Usuário)
    \item[IA] Inteligência Artificial
    \item[ID] Identificação
    \item[IoT] \textit{Internet of Things} (Internet das Coisas)
    \item[IP] \textit{Internet Protocol} (Protocolo de Internet)
    \item[IQR] \textit{Interquartile Range} (Intervalo Interquartil)
    \item[ISO] \textit{International Organization for Standardization} (Organização Internacional para Padronização)
    \item[KNN] \textit{K-Nearest Neighbors} (K-Vizinhos Mais Próximos)
    \item[LSTM] \textit{Long Short-Term Memory} (Memória de Curto-Longo Prazo)

    \item[PDF] \textit{Portable Document Format} (Formato de Documento Portátil)
    \item[RAM] \textit{Random Access Memory} (Memória de Acesso Aleatório)
    \item[RMS] \textit{Root Mean Square} (Valor Quadrático Médio)
    \item[RPM] Rotações Por Minuto
    \item[SQL] \textit{Structured Query Language} (Linguagem de Consulta Estruturada)
    \item[UTC] \textit{Universal Time Coordinated} (Tempo Universal Coordenado)
\end{description}
